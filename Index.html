// Core logic refactored for gas optimization // SPDX-License-Identifier: MIT pragma solidity 0.8.24; contract ChronoVault { struct LockBox { uint256 balance; uint256 unlockTimestamp; string purpose; bool exists; } address public immutable vaultManager; uint256 public totalVaultDeposits; mapping(address => LockBox) private lockBoxes; event Deposited(address indexed user, uint256 amount, uint256 unlockTime, string purpose); event Withdrawn(address indexed user, uint256 amount); event TimeExtended(address indexed user, uint256 newUnlockTime); modifier onlyManager() { require(msg.sender == vaultManager, "ChronoVault: Unauthorized"); _; } constructor() { vaultManager = msg.sender; } // Создать или пополнить сейф с блокировкой по времени (в секундах) function deposit(uint256 lockDurationSeconds, string calldata purpose) external payable { require(msg.value > 0, "ChronoVault: Zero deposit rejected"); require(lockDurationSeconds >= 60, "ChronoVault: Min lock duration is 60 seconds"); LockBox storage box = lockBoxes[msg.sender]; uint256 newUnlockTime = block.timestamp + lockDurationSeconds; if (box.exists) { box.balance += msg.value; if (newUnlockTime > box.unlockTimestamp) { box.unlockTimestamp = newUnlockTime; } } else { lockBoxes[msg.sender] = LockBox({ balance: msg.value, unlockTimestamp: newUnlockTime, purpose: purpose, exists: true }); } totalVaultDeposits += msg.value; emit Deposited(msg.sender, msg.value, box.unlockTimestamp, purpose); } // Вывести средства после наступления даты разблокировки function withdraw() external { LockBox storage box = lockBoxes[msg.sender]; require(box.exists, "ChronoVault: No deposit found"); require(block.timestamp >= box.unlockTimestamp, "ChronoVault: Funds still locked"); require(box.balance > 0, "ChronoVault: Empty balance"); uint256 amount = box.balance; box.balance = 0; totalVaultDeposits -= amount; (bool success, ) = payable(msg.sender).call{value: amount}(""); require(success, "ChronoVault: Transfer failed"); emit Withdrawn(msg.sender, amount); } // Продлить блокировку function extendLock(uint256 extraSeconds) external { LockBox storage box = lockBoxes[msg.sender]; require(box.exists, "ChronoVault: Deposit not active"); require(extraSeconds > 0, "ChronoVault: Invalid extension"); box.unlockTimestamp += extraSeconds; emit TimeExtended(msg.sender, box.unlockTimestamp); } // Просмотр информации о сейфе function getBoxInfo(address user) external view returns ( uint256 balance, uint256 unlockTimestamp, uint256 secondsRemaining, string memory purpose ) { LockBox memory box = lockBoxes[user]; if (!box.exists) return (0, 0, 0, ""); uint256 remaining = 0; if (box.unlockTimestamp > block.timestamp) { remaining = box.unlockTimestamp - block.timestamp; } return (box.balance, box.unlockTimestamp, remaining, box.purpose); } }
